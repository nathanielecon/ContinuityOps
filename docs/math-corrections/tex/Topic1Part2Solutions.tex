\lesson{Lesson 1-1: Relate Integers and Opposites}

\question{Question 1}
\need{The sum of a $24$ m descent and a $24$ m climb (Part A), and why the two changes are additive inverses (Part B).}
\know{Descending is negative and climbing is positive. Additive inverses have the same size and opposite signs, so they add to zero.}
\solve{$(-24) + (+24) = 0$}
\ans{Part A: $(-24) + 24 = 0$; Part B: equal magnitude and opposite signs, so the two changes cancel}

\question{Question 2}
\need{An equation with $\abs{p}$ and $\abs{q}$.}
\know{$p + q = 0$ means $q = -p$, so the distances from zero match.}
\solve{$\abs{p} = \abs{q}$}
\ans{$\abs{p} = \abs{q}$ (or equivalent)}

\question{Question 3}
\need{$\abs{-11}$}
\know{Absolute value is distance from zero.}
\solve{$\abs{-11} = 11$}
\ans{$11$}

\question{Question 4}
\need{Order $-2$, $4$, $6$ least to greatest.}
\know{On a number line, more left means smaller.}
\solve{$-2$, $4$, $6$}
\ans{$-2$, $4$, $6$}

\lesson{Lesson 1-2: Understand Rational Numbers}

\question{Question 1}
\need{Decimal for $\tfrac{5}{8}$.}
\know{Divide $5 \div 8$.}
\solve{$5 \div 8 = 0.625$}
\ans{$0.625$}

\question{Question 2}
\need{Which statement is \emph{always} right.}
\know{A fraction (with nonzero denominator) becomes a decimal that either terminates or repeats.}
\solve{B covers both possibilities, so B is always right. A fails for $\tfrac13$ and C fails for $\tfrac12$.}
\ans{B}

\question{Question 3}
\need{Who is correct about $0.\overline{6}$, Mia or Lee.}
\know{Let $x = 0.\overline{6}$. Then $10x - x = 9x$, and the repeating tails cancel.}
\solve{$10x = 6.\overline{6}$, so $9x = 6$ and $x = \tfrac{6}{9} = \tfrac{2}{3}$. A repeating decimal is rational.}
\ans{Mia; $0.\overline{6} = \tfrac{2}{3}$}

\lesson{Lesson SC-1: Compare Rational Numbers}

\question{Question 1}
\need{Two comparisons of $9.3$ and $9.8$.}
\know{On a number line, more right means greater.}
\solve{$9.3 < 9.8$ and $9.8 > 9.3$}
\ans{$9.3 < 9.8$ and $9.8 > 9.3$ (pounds OK)}

\question{Question 2}
\need{Symbols that fit $\tfrac{2}{5}$ and $0.4$.}
\know{$\tfrac{2}{5} = 0.4$}
\solve{$\tfrac{2}{5} = 0.4$ exactly, so of the four symbols offered ($<$, $>$, $=$, $\neq$) only $=$ is true.}

\lesson{Lesson 1-4: Subtract Integers}

\need{Distance from $12\degF$ to $3\degF$.}
\know{Distance is the absolute value of the difference.}
\solve{$\abs{12-3} = 9$. Valid: $\abs{12-3}$, $\abs{3-12}$, $12-3$.}

\question{Question 2}

\question{Question 3}

\lesson{Lesson 1-5: Add and Subtract Rational Numbers}

\question{Question 1}
\need{Balance after 5 months.}
\know{Start $\$180$ (below $\$200$), so the fee applies each month.}
\solve{$180 - 12 = 168$; $168 - 12 = 156$; $156 - 12 = 144$; $144 - 12 = 132$; $132 - 12 = 120$.}
\ans{$\$120$}

\question{Question 2}
\need{Temperature after the two drops.}
\know{Each drop is a negative change; add them to the start.}
\solve{$-4.6 - 1.35 - 2.05 = -8.00$}
\ans{$-8\degC$}

\question{Question 3}
\need{Vertical distance between the tower and the canyon floor.}
\know{Distance is the absolute value of the difference; write the mixed number as a decimal first.}
\solve{$3\tfrac{1}{4} = 3.25$; $\abs{3.25 - (-2.5)} = 5.75$.}
\ans{$5.75$ miles}

\lesson{Lesson 1-6: Multiply Integers}

\question{Question 1}
\parta{Ascent is positive: $20$ (or $+20$) feet per minute.}
\partb{$20 \times 9 = 180$ feet above sea level.}
\ans{Part A: $20$ (or $+20$); Part B: $180$ feet above sea level}

\question{Question 2}
\need{Height after $3.5$ minutes.}
\know{An ascent rate is positive, so the product is positive.}
\solve{$180.25 \times 3.5 = 630.875$ m $= 0.630875$ km $\approx 1$ km.}
\ans{$1$ kilometer (nearest km; ascent)}

\question{Question 3}
\need{$(-6)^{2}$}
\know{The parentheses mean the negative is part of the base.}
\solve{$(-6)^{2} = (-6) \times (-6) = 36$}
\ans{$36$}

\question{Question 4}
\need{$-6^{2}$}
\know{Without parentheses, the exponent applies to $6$ only.}
\solve{$-6^{2} = -(6^{2}) = -36$}
\ans{$-36$}
\difference{parentheses square the negative; without them, square first, then make negative.}

\lesson{Lesson 1-8: Divide Integers}

\question{Question 1}
\parta{Unit rate $= 140 \div 5 = 28$ feet per minute.}
\partb{$28 \times 3 = 84$ feet above sea level.}
\ans{Part A: $28$ (or $+28$); Part B: $84$ feet above sea level}

\question{Question 2}
\need{The quotient.}
\know{Opposite signs $\Rightarrow$ negative quotient.}
\solve{$(-81) \div 9 = -9$}
\ans{$-9$}

\question{Question 3}
\need{The quotient.}
\know{Same signs $\Rightarrow$ positive quotient.}
\solve{$(-56) \div (-7) = 8$}
\ans{$8$}

\question{Question 4}
\need{$n$ in $\dfrac{-24}{8} = \dfrac{n}{-3}$.}
\know{The fractions are equal. Use the butterfly method.}
\butterfly{-24}{8}{n}{-3}
\solve{Cross products: $(-24)(-3) = 8 \cdot n$, so $72 = 8n$ and $n = 9$.}
\checkline{$(-24) \div 8 = -3$ and $9 \div (-3) = -3$.}
\ans{$n = 9$}

\lesson{Lesson 1-9: Divide Rational Numbers}

\question{Question 1}
\need{Each owner's share of the loss.}
\know{Split the loss equally among the 5 owners.}
\solve{$(-625.50) \div 5 = -125.10$}
\ans{$-125.10$ (or $-\$125.10$ per owner)}

\question{Question 2}
\need{How many rest stops.}
\know{$\mixed{16}{1}{2} = \tfrac{33}{2}$, $\mixed{2}{3}{4} = \tfrac{11}{4}$.}
\solve{$\tfrac{33}{2} \div \tfrac{11}{4} = \tfrac{33}{2} \times \tfrac{4}{11} = 6$ intervals. Rest stops at the end of each interval (including the finish line) $\Rightarrow 6$.}
\ans{$6$ rest stops}

\lesson{Lesson SC-2: Use the Laws of Exponents}

\question{Question 1}

\question{Question 4}

\question{Question 8}

\question{Question 9}

\question{Question 10}
